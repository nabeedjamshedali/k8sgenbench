          ports:
            - name: mysql
              containerPort: 3306
          env:
            - name: MYSQL_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: catalogue-db-secret
                  key: mysql-root-password
            - name: MYSQL_DATABASE
              valueFrom:
                configMapKeyRef:
                  name: sock-shop-config
                  key: MYSQL_DATABASE

              mountPath: /var/lib/mysql

            requests:
              cpu: 100m
              memory: 256Mi

              memory: 1Gi

            periodSeconds: 5
            failureThreshold: 30
          readinessProbe:
            tcpSocket:
              port: mysql
            periodSeconds: 10

            tcpSocket:
              port: mysql
            periodSeconds: 20
            failureThreshold: 5

        name: catalogue-data
