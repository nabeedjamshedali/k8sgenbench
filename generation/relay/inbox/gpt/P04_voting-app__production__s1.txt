# ============================================================
# Example Voting App - Production Kubernetes Manifests
# ============================================================
# Components:
#   - Namespace
#   - PostgreSQL Secret
#   - PostgreSQL StatefulSet + Persistent Storage
#   - Redis StatefulSet + Persistent Storage
#   - Vote Deployment + Service
#   - Result Deployment + Service
#   - Worker Deployment
#   - Ingress
#   - Network Policies
#   - Pod Disruption Budgets
#
# Prerequisites:
#   - Kubernetes cluster with a default StorageClass
#   - NGINX Ingress Controller
#   - TLS certificate stored in Secret voting-app-tls
#   - Replace example.com domains with actual domains
#   - Replace PostgreSQL password before deployment
#   - Pin container images to tested immutable digests
#
# Deploy:
#   kubectl apply -f voting-app.yaml
# ============================================================

apiVersion: v1
kind: Namespace
metadata:
  name: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
    environment: production
    kubernetes.io/metadata.name: voting-app

---
# ============================================================
# PostgreSQL Credentials
# ============================================================

apiVersion: v1
kind: Secret
metadata:
  name: postgres-secret
  namespace: voting-app
  labels:
    app: db
type: Opaque
stringData:
  POSTGRES_USER: "postgres"
  POSTGRES_PASSWORD: "REPLACE_WITH_STRONG_PASSWORD"
  POSTGRES_DB: "postgres"

---
# ============================================================
# Redis Service
# ============================================================

apiVersion: v1
kind: Service
metadata:
  name: redis
  namespace: voting-app
  labels:
    app: redis
spec:
  type: ClusterIP
  selector:
    app: redis
  ports:
    - name: redis
      port: 6379
      targetPort: redis
      protocol: TCP

---
# ============================================================
# Redis StatefulSet
# ============================================================

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: redis
  namespace: voting-app
  labels:
    app: redis
spec:
  serviceName: redis
  replicas: 1
  revisionHistoryLimit: 5

  selector:
    matchLabels:
      app: redis

  template:
    metadata:
      labels:
        app: redis
        tier: backend

    spec:
      terminationGracePeriodSeconds: 30

      securityContext:
        runAsNonRoot: true
        runAsUser: 999
        runAsGroup: 999
        fsGroup: 999
        seccompProfile:
          type: RuntimeDefault

      containers:
        - name: redis
          image: redis:alpine
          imagePullPolicy: IfNotPresent

          command:
            - redis-server

          args:
            - "--appendonly"
            - "yes"
            - "--appendfsync"
            - "everysec"
            - "--maxmemory"
            - "256mb"
            - "--maxmemory-policy"
            - "noeviction"

          ports:
            - name: redis
              containerPort: 6379
              protocol: TCP

          resources:
            requests:
              cpu: "100m"
              memory: "128Mi"
            limits:
              cpu: "500m"
              memory: "512Mi"

          startupProbe:
            exec:
              command:
                - redis-cli
                - ping
            periodSeconds: 5
            failureThreshold: 12

          readinessProbe:
            exec:
              command:
                - redis-cli
                - ping
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 3

          livenessProbe:
            exec:
              command:
                - redis-cli
                - ping
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 3

          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
                - ALL

          volumeMounts:
            - name: redis-data
              mountPath: /data

  volumeClaimTemplates:
    - metadata:
        name: redis-data
        labels:
          app: redis
      spec:
        accessModes:
          - ReadWriteOnce
        resources:
          requests:
            storage: 5Gi

---

  name: db
  namespace: voting-app
  labels:
    app: db
spec:
  serviceName: db
  replicas: 1
  revisionHistoryLimit: 5

  selector:
    matchLabels:
      app: db

  template:
    metadata:
      labels:
        app: db
        tier: backend

    spec:
      terminationGracePeriodSeconds: 60

      securityContext:
        fsGroup: 70
        seccompProfile:
          type: RuntimeDefault

      containers:
        - name: postgres
          image: postgres:15-alpine
          imagePullPolicy: IfNotPresent

          ports:
            - name: postgres
              containerPort: 5432
              protocol: TCP

          envFrom:
            - secretRef:
                name: postgres-secret

          env:
            - name: PGDATA
              value: /var/lib/postgresql/data/pgdata

          resources:
            requests:
              cpu: "250m"
              memory: "256Mi"
            limits:
              cpu: "1000m"
              memory: "1Gi"

          startupProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 30

          readinessProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 3

          livenessProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 6

          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
                - ALL
              add:
                - CHOWN
                - DAC_OVERRIDE
                - FOWNER
                - SETGID
                - SETUID

          volumeMounts:
            - name: postgres-data
              mountPath: /var/lib/postgresql/data

  volumeClaimTemplates:
    - metadata:
        name: postgres-data
        labels:
          app: db
      spec:
        accessModes:
          - ReadWriteOnce
        resources:
          requests:
            storage: 10Gi

# ============================================================
# Ingress
# ============================================================
# Requires NGINX Ingress Controller.
#
# DNS:
#   vote.example.com   -> Ingress Controller
#   result.example.com -> Ingress Controller
#
# Create the voting-app-tls Kubernetes TLS Secret separately.
# ============================================================

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: voting-app-ingress
  namespace: voting-app

  labels:
    app: voting-app

  annotations:
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
    nginx.ingress.kubernetes.io/proxy-connect-timeout: "10"
    nginx.ingress.kubernetes.io/proxy-read-timeout: "3600"
    nginx.ingress.kubernetes.io/proxy-send-timeout: "3600"
    nginx.ingress.kubernetes.io/proxy-buffering: "off"

spec:
  ingressClassName: nginx

  tls:
    - hosts:
        - vote.example.com
        - result.example.com
      secretName: voting-app-tls

  rules:
    - host: vote.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: vote
                port:
                  number: 80

    - host: result.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: result
                port:
                  number: 80

---

# Production Deployment Notes
# ============================================================
#
# 1. Replace the placeholder PostgreSQL password and use a
#    secure external secret-management system.
#
# 2. Pin all application and infrastructure images to tested
#    immutable digests before production rollout.
#
# 3. Replace example.com domains with actual DNS names.
#
# 4. Provision the voting-app-tls Secret using cert-manager
#    or another trusted certificate-management solution.
#
# 5. Ensure the cluster has:
#      - NGINX Ingress Controller
#      - metrics-server
#      - NetworkPolicy-capable CNI
#      - Default StorageClass
#
# 6. Configure regular PostgreSQL backups, restore testing,
#    PVC snapshots, and off-cluster backup retention.
#
# 7. Redis and PostgreSQL run as single instances.
#    For high availability, use managed services or
#    appropriately configured database operators.
#
# 8. The original application is a demonstration project.
#    Validate multi-replica application behavior, session
#    handling, and database initialization before production.
#
# 9. The Compose seed profile is intentionally not deployed.
#    Seeding should use a separate, explicitly triggered Job
#    with a published seed-data image.
#
# 10. Verify the health, security and compatibility of the
