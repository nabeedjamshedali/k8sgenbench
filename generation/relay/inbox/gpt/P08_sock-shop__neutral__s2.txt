  replicas: 1

    type: Recreate

      app: catalogue-db

      labels:
        app: catalogue-db
    spec:
      hostname: catalogue-db
      containers:
        - name: catalogue-db
          image: weaveworksdemos/catalogue-db:0.3.0
          ports:
            - containerPort: 3306
          env:
            - name: MYSQL_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: catalogue-db-secret
                  key: MYSQL_ROOT_PASSWORD

              value: "true"
            - name: MYSQL_DATABASE
              value: "socksdb"
          volumeMounts:
            - name: data
              mountPath: /var/lib/mysql
          resources:
            requests:
              cpu: 100m
              memory: 256Mi
            limits:
              cpu: "1"
              memory: 1Gi
      volumes:
        - name: data
          persistentVolumeClaim:
            claimName: catalogue-db-data
