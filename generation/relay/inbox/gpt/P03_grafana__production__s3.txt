# Production-oriented, single-replica Grafana deployment.
# Before applying:
# - Replace the Secret placeholders with securely generated values.
# - Configure grafana.example.com DNS and the TLS certificate.
# - Ensure the cluster has a default StorageClass.
# - Store production secrets in an external secret manager.
# - SQLite with one replica is used; for HA, use external PostgreSQL.

apiVersion: v1
kind: Namespace
metadata:
  name: my-grafana
  labels:
    app.kubernetes.io/part-of: grafana
    pod-security.kubernetes.io/enforce: restricted

---
apiVersion: v1
kind: Secret
metadata:
  name: grafana-credentials
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
type: Opaque
stringData:
  admin-user: "admin"
  admin-password: "REPLACE_WITH_STRONG_RANDOM_PASSWORD"
  secret-key: "REPLACE_WITH_LONG_RANDOM_ENCRYPTION_KEY"

---
apiVersion: v1
kind: ConfigMap
metadata:
  name: grafana-config
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
data:
  grafana.ini: |
    app_mode = production

    [paths]
    data = /var/lib/grafana
    logs = /var/log/grafana
    plugins = /var/lib/grafana/plugins
    provisioning = /etc/grafana/provisioning

    [server]
    protocol = http
    http_addr = 0.0.0.0
    http_port = 3000
    domain = grafana.example.com
    root_url = https://grafana.example.com/
    enforce_domain = true
    enable_gzip = true
    router_logging = false

    [database]
    type = sqlite3
    path = grafana.db
    wal = true
    high_availability = false

    [security]
    disable_initial_admin_creation = false
    cookie_secure = true
    cookie_samesite = strict
    allow_embedding = false
    disable_gravatar = true
    disable_brute_force_login_protection = false
    disable_username_login_protection = false
    disable_ip_address_login_protection = false
    x_content_type_options = true
    strict_transport_security = true
    strict_transport_security_max_age_seconds = 86400

    [users]
    allow_sign_up = false
    allow_org_create = false
    auto_assign_org = true
    auto_assign_org_role = Viewer
    default_theme = dark

    [auth]
    disable_login = false
    disable_login_form = false
    token_rotation_interval_minutes = 10

    [auth.anonymous]
    enabled = false

    [auth.basic]
    enabled = true
    password_policy = true

    [analytics]
    reporting_enabled = false
    check_for_updates = true
    check_for_plugin_updates = true

    [snapshots]
    enabled = true
    external_enabled = false
    public_mode = false

    [dashboards]
    versions_to_keep = 20
    min_refresh_interval = 5s

    [log]
    mode = console
    level = info

    [log.console]
    format = json

    [metrics]
    enabled = true
    disable_total_stats = false

    [smtp]
    enabled = false

---
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: grafana-storage
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 20Gi

---
apiVersion: v1
kind: ServiceAccount
metadata:
  name: grafana
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
automountServiceAccountToken: false

---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: grafana
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
    app.kubernetes.io/component: server
spec:
  replicas: 1
  revisionHistoryLimit: 5
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app.kubernetes.io/name: grafana
      app.kubernetes.io/component: server
  template:
    metadata:
      labels:
        app.kubernetes.io/name: grafana
        app.kubernetes.io/component: server
      annotations:
        prometheus.io/scrape: "true"
        prometheus.io/path: "/metrics"
        prometheus.io/port: "3000"
    spec:
      serviceAccountName: grafana
      automountServiceAccountToken: false
      terminationGracePeriodSeconds: 60

      securityContext:
        runAsNonRoot: true
        runAsUser: 472
        runAsGroup: 472
        fsGroup: 472
        fsGroupChangePolicy: OnRootMismatch
        seccompProfile:
          type: RuntimeDefault

      containers:
        - name: grafana
          image: grafana/grafana:12.1.0
          imagePullPolicy: IfNotPresent

          ports:
            - name: http
              containerPort: 3000
              protocol: TCP

          env:
            - name: GF_PATHS_CONFIG
              value: /etc/grafana/grafana.ini

            - name: GF_PATHS_DATA
              value: /var/lib/grafana

            - name: GF_PATHS_LOGS
              value: /var/log/grafana

            - name: GF_PATHS_PLUGINS
              value: /var/lib/grafana/plugins

            - name: GF_SECURITY_ADMIN_USER
              valueFrom:
                secretKeyRef:
                  name: grafana-credentials
                  key: admin-user

            - name: GF_SECURITY_ADMIN_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: grafana-credentials
                  key: admin-password

            - name: GF_SECURITY_SECRET_KEY
              valueFrom:
                secretKeyRef:
                  name: grafana-credentials
                  key: secret-key

          resources:
            requests:
              cpu: 250m
              memory: 512Mi
              ephemeral-storage: 256Mi
            limits:
              cpu: "2"
              memory: 2Gi
              ephemeral-storage: 2Gi

          startupProbe:
            httpGet:
              path: /api/health
              port: http
              scheme: HTTP
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 30

          readinessProbe:
            httpGet:
              path: /api/health
              port: http
              scheme: HTTP
            initialDelaySeconds: 5
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 3

          livenessProbe:
            httpGet:
              path: /api/health
              port: http
              scheme: HTTP
            periodSeconds: 30
            timeoutSeconds: 5
            failureThreshold: 3

          securityContext:
            allowPrivilegeEscalation: false
            readOnlyRootFilesystem: true
            capabilities:
              drop:
                - ALL

          volumeMounts:
            - name: grafana-config
              mountPath: /etc/grafana/grafana.ini
              subPath: grafana.ini
              readOnly: true

            - name: grafana-storage
              mountPath: /var/lib/grafana

            - name: grafana-logs
              mountPath: /var/log/grafana

            - name: grafana-tmp
              mountPath: /tmp

      volumes:
        - name: grafana-config
          configMap:
            name: grafana-config
            defaultMode: 0444

        - name: grafana-storage
          persistentVolumeClaim:
            claimName: grafana-storage

        - name: grafana-logs
          emptyDir:
            sizeLimit: 1Gi

        - name: grafana-tmp
          emptyDir:
            sizeLimit: 512Mi

---
apiVersion: v1
kind: Service
metadata:
  name: grafana
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
    app.kubernetes.io/component: server
spec:
  type: ClusterIP
  selector:
    app.kubernetes.io/name: grafana
    app.kubernetes.io/component: server
  ports:
    - name: http
      port: 3000
      targetPort: http
      protocol: TCP
  sessionAffinity: None

---
# Requires an installed NGINX Ingress Controller.
# Create the grafana-tls Secret with a valid certificate
# for grafana.example.com before enabling external access.
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: grafana
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
  annotations:
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
    nginx.ingress.kubernetes.io/force-ssl-redirect: "true"
    nginx.ingress.kubernetes.io/proxy-body-size: "10m"
    nginx.ingress.kubernetes.io/proxy-read-timeout: "60"
    nginx.ingress.kubernetes.io/proxy-send-timeout: "60"
spec:
  ingressClassName: nginx
  tls:
    - hosts:
        - grafana.example.com
      secretName: grafana-tls
  rules:
    - host: grafana.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: grafana
                port:
                  name: http

---
# Allows Grafana traffic from the ingress controller
# and monitoring workloads. Adapt namespace labels
# to match the ingress controller installation.
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: grafana-network-policy
  namespace: my-grafana
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/name: grafana
      app.kubernetes.io/component: server

  policyTypes:
    - Ingress
    - Egress

  ingress:
    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: ingress-nginx
      ports:
        - protocol: TCP
          port: 3000

    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: monitoring
      ports:
        - protocol: TCP
          port: 3000

  egress:
    # DNS resolution
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53

    # HTTPS for external data sources and Grafana services.
    # Restrict destinations for the actual environment.
    - ports:
        - protocol: TCP
          port: 443

    # HTTP for internal data sources, where required.
    # Restrict destinations for the actual environment.
    - ports:
        - protocol: TCP
          port: 80

    # Common Prometheus endpoint port.
    - ports:
        - protocol: TCP
          port: 9090

---
# Grafana runs as a single replica using SQLite.
# This budget permits voluntary eviction, but does
# not provide application high availability.
apiVersion: policy/v1
kind: PodDisruptionBudget
metadata:
  name: grafana-pdb
  namespace: my-grafana
  labels:
    app.kubernetes.io/name: grafana
spec:
  minAvailable: 0
  selector:
    matchLabels:
      app.kubernetes.io/name: grafana
      app.kubernetes.io/component: server
