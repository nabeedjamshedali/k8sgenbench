# ============================================================
# PostgreSQL Initialization
# Creates the n8n application database role with privileges.
# Executed only when PostgreSQL initializes an empty data dir.
# ============================================================
apiVersion: v1
kind: ConfigMap
metadata:
  name: postgres-init
  namespace: n8n
data:
  init-data.sh: |
    #!/bin/bash
    set -Eeuo pipefail

    if [[ -z "${POSTGRES_NON_ROOT_USER:-}" ]] ||
       [[ -z "${POSTGRES_NON_ROOT_PASSWORD:-}" ]]; then
      echo "ERROR: Application database credentials are required."
      exit 1
    fi

    if [[ "$POSTGRES_NON_ROOT_USER" == "$POSTGRES_USER" ]]; then
      echo "ERROR: Application user must differ from admin user."
      exit 1
    fi

    export PGUSER="$POSTGRES_USER"
    export PGDATABASE="$POSTGRES_DB"

    psql \
      -X \
      -v ON_ERROR_STOP=1 \
      -v app_user="$POSTGRES_NON_ROOT_USER" \
      -v app_password="$POSTGRES_NON_ROOT_PASSWORD" \
      -v app_database="$POSTGRES_DB" <<'EOSQL'

    SELECT format(
      'CREATE ROLE %I LOGIN PASSWORD %L',
      :'app_user',
      :'app_password'
    )
    WHERE NOT EXISTS (
      SELECT 1 FROM pg_roles
      WHERE rolname = :'app_user'
    )
    \gexec

    SELECT format(
      'GRANT CONNECT, TEMPORARY, CREATE ON DATABASE %I TO %I',
      :'app_database',
      :'app_user'
    )
    \gexec

    SELECT format(
      'GRANT USAGE, CREATE ON SCHEMA public TO %I',
      :'app_user'
    )
    \gexec

    EOSQL

    echo "PostgreSQL application user initialization completed."

# ============================================================
# PostgreSQL StatefulSet
#
# Uses PostgreSQL 18's versioned data directory layout.
# The PVC is mounted at /var/lib/postgresql.
#
# For existing installations, migrate the data rather than
# attaching an incompatible PostgreSQL 16 data directory.
# ============================================================
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: postgres
  namespace: n8n
  labels:
    app: postgres
spec:
  serviceName: postgres
  replicas: 1

  revisionHistoryLimit: 3

  persistentVolumeClaimRetentionPolicy:
    whenDeleted: Retain
    whenScaled: Retain

  selector:
    matchLabels:
      app: postgres

  template:
    metadata:
      labels:
        app: postgres
    spec:
      terminationGracePeriodSeconds: 120

      securityContext:
        runAsUser: 999
        runAsGroup: 999
        fsGroup: 999
        seccompProfile:
          type: RuntimeDefault

      containers:
        - name: postgres
          image: postgres:18
          imagePullPolicy: IfNotPresent

          ports:
            - name: postgres
              containerPort: 5432
              protocol: TCP

          env:
            - name: POSTGRES_USER
              valueFrom:
                secretKeyRef:
                  name: n8n-secrets
                  key: POSTGRES_USER

            - name: POSTGRES_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: n8n-secrets
                  key: POSTGRES_PASSWORD

            - name: POSTGRES_DB
              valueFrom:
                configMapKeyRef:
                  name: n8n-config
                  key: POSTGRES_DB

            - name: POSTGRES_NON_ROOT_USER
              valueFrom:
                secretKeyRef:
                  name: n8n-secrets
                  key: POSTGRES_NON_ROOT_USER

            - name: POSTGRES_NON_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: n8n-secrets
                  key: POSTGRES_NON_ROOT_PASSWORD

            - name: PGDATA
              value: "/var/lib/postgresql/18/docker"

            - name: PGHOST
              value: "/var/run/postgresql"

          resources:
            requests:
              cpu: "250m"
              memory: "512Mi"
            limits:
              cpu: "2000m"
              memory: "2Gi"

          startupProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -h 127.0.0.1 -p 5432 \
                    -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 5
            timeoutSeconds: 5
            failureThreshold: 60

          readinessProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -h 127.0.0.1 -p 5432 \
                    -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 3

          livenessProbe:
            exec:
              command:
                - /bin/sh
                - -ec
                - |
                  pg_isready -h 127.0.0.1 -p 5432 \
                    -U "$POSTGRES_USER" -d "$POSTGRES_DB"
            periodSeconds: 20
            timeoutSeconds: 5
            failureThreshold: 6

          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
                - ALL
            seccompProfile:
              type: RuntimeDefault

          volumeMounts:
            - name: postgres-data
              mountPath: /var/lib/postgresql

            - name: postgres-init
              mountPath: /docker-entrypoint-initdb.d/init-data.sh
              subPath: init-data.sh
              readOnly: true

            - name: postgres-run
              mountPath: /var/run/postgresql

      volumes:
        - name: postgres-init
          configMap:
            name: postgres-init
            defaultMode: 0555

        - name: postgres-run
          emptyDir: {}

  volumeClaimTemplates:
    - metadata:
        name: postgres-data
        labels:
          app: postgres
      spec:
        accessModes:
          - ReadWriteOnce

        # have an appropriate default StorageClass.
        resources:
          requests:
            storage: 20Gi

# ============================================================
# Public HTTPS Ingress
#
# Requires an ingress-nginx controller and a pre-existing
# Kubernetes TLS Secret named n8n-tls in namespace n8n.
#
# Replace n8n.example.com with your domain.
# Point its DNS record to the Ingress controller.
#
# Only port 5678 is exposed.
# The runner broker port 5679 remains internal.
# ============================================================
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: n8n
  namespace: n8n
  annotations:
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
    nginx.ingress.kubernetes.io/force-ssl-redirect: "true"
    nginx.ingress.kubernetes.io/proxy-body-size: "50m"
    nginx.ingress.kubernetes.io/proxy-read-timeout: "300"
    nginx.ingress.kubernetes.io/proxy-send-timeout: "300"
spec:
  ingressClassName: nginx

  tls:
    - hosts:
        - n8n.example.com
      secretName: n8n-tls

  rules:
    - host: n8n.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: n8n
                port:
                  name: http
