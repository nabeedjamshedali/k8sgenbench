# Replace all placeholder credentials before deployment.
# POSTGRES_USER and POSTGRES_NON_ROOT_USER must be different.
apiVersion: v1
kind: Secret
metadata:
  name: n8n-secrets
  namespace: n8n
type: Opaque
stringData:
  POSTGRES_USER: "postgres"
  POSTGRES_PASSWORD: "REPLACE_WITH_STRONG_POSTGRES_PASSWORD"
  POSTGRES_DB: "n8n"
  POSTGRES_NON_ROOT_USER: "n8n_app"
  POSTGRES_NON_ROOT_PASSWORD: "REPLACE_WITH_STRONG_APP_PASSWORD"
  RUNNERS_AUTH_TOKEN: "REPLACE_WITH_STRONG_RUNNER_AUTH_TOKEN"
  N8N_ENCRYPTION_KEY: "REPLACE_WITH_PERSISTENT_RANDOM_ENCRYPTION_KEY"
---
apiVersion: v1
kind: ConfigMap
metadata:
  name: postgres-init
  namespace: n8n
data:
  init-data.sh: |
    #!/bin/bash
    set -euo pipefail

    if [ -n "${POSTGRES_NON_ROOT_USER:-}" ] && \
       [ -n "${POSTGRES_NON_ROOT_PASSWORD:-}" ]; then

      psql -v ON_ERROR_STOP=1 \
        --username "$POSTGRES_USER" \
        --dbname "$POSTGRES_DB" \
        -v app_user="$POSTGRES_NON_ROOT_USER" \
        -v app_password="$POSTGRES_NON_ROOT_PASSWORD" \
        -v db_name="$POSTGRES_DB" <<'EOSQL'

    SELECT format(
      'CREATE ROLE %I LOGIN PASSWORD %L',
      :'app_user',
      :'app_password'
    )
    WHERE NOT EXISTS (
      SELECT 1 FROM pg_roles
      WHERE rolname = :'app_user'
    ) \gexec

    SELECT format(
      'GRANT ALL PRIVILEGES ON DATABASE %I TO %I',
      :'db_name',
      :'app_user'
    ) \gexec

    SELECT format(
      'GRANT USAGE, CREATE ON SCHEMA public TO %I',
      :'app_user'
    ) \gexec

    EOSQL

    else
      echo "SETUP INFO: No non-root PostgreSQL credentials provided!"
      exit 1
    fi
---

apiVersion: apps/v1
kind: Deployment
metadata:
  name: n8n-runner
  namespace: n8n
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app: n8n-runner
  template:
    metadata:
      labels:
        app: n8n-runner
    spec:
      terminationGracePeriodSeconds: 30
      initContainers:
        - name: wait-for-n8n
          image: busybox:1.36
          imagePullPolicy: IfNotPresent
          command:
            - /bin/sh
            - -c
            - |
              until wget -q -O /dev/null \
                http://n8n:5678/healthz; do
                echo "Waiting for n8n..."
                sleep 5
              done
              echo "n8n is ready."
          resources:
            requests:
              cpu: "25m"
              memory: "32Mi"

              cpu: "100m"
              memory: "128Mi"
