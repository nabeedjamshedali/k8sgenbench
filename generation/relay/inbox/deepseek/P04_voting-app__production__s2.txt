apiVersion: v1

apiVersion: apps/v1

  namespace: voting-app

    app: redis

  replicas: 1

    matchLabels:
      app: redis
  template:
    metadata:
      labels:
        app: redis
    spec:
      containers:
        - name: redis
          image: redis:alpine
          ports:
            - containerPort: 6379
          volumeMounts:
            - name: healthchecks
              mountPath: /healthchecks
          readinessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/redis.sh"]
            initialDelaySeconds: 5
            periodSeconds: 5
          livenessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/redis.sh"]
            initialDelaySeconds: 15
            periodSeconds: 10
          resources:
            requests:

              memory: 64Mi
            limits:
              cpu: 250m
              memory: 128Mi
      volumes:

            name: redis-healthcheck

              mountPath: /healthchecks

            exec:
              command: ["/bin/sh", "/healthchecks/postgres.sh"]
            initialDelaySeconds: 5
            periodSeconds: 5
          livenessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/postgres.sh"]
            initialDelaySeconds: 15
            periodSeconds: 10

              memory: 256Mi

              cpu: 500m
              memory: 512Mi

            claimName: db-data

            name: postgres-healthcheck
