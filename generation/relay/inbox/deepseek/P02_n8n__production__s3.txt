# -----------------------------------------------------------------------------
# ConfigMap: Postgres init script (from init-data.sh)
# The docker-entrypoint-initdb.d scripts run once on an empty data directory.
# -----------------------------------------------------------------------------
apiVersion: v1
kind: ConfigMap
metadata:
  name: postgres-initdb
  namespace: n8n
  labels:
    app.kubernetes.io/name: postgres
    app.kubernetes.io/part-of: n8n-stack
data:
  init-data.sh: |
    #!/bin/bash
    set -e;

    if [ -n "${POSTGRES_NON_ROOT_USER:-}" ] && [ -n "${POSTGRES_NON_ROOT_PASSWORD:-}" ]; then
        psql -v ON_ERROR_STOP=1 --username "$POSTGRES_USER" --dbname "$POSTGRES_DB" <<-EOSQL
            CREATE USER ${POSTGRES_NON_ROOT_USER} WITH PASSWORD '${POSTGRES_NON_ROOT_PASSWORD}';
            GRANT ALL PRIVILEGES ON DATABASE ${POSTGRES_DB} TO ${POSTGRES_NON_ROOT_USER};
            GRANT CREATE ON SCHEMA public TO ${POSTGRES_NON_ROOT_USER};
    EOSQL
    else
        echo "SETUP INFO: No Environment variables given!"
    fi

# -----------------------------------------------------------------------------
# PostgreSQL: Deployment
# -----------------------------------------------------------------------------
apiVersion: apps/v1
kind: Deployment
metadata:
  name: postgres
  namespace: n8n
  labels:
    app.kubernetes.io/name: postgres
    app.kubernetes.io/part-of: n8n-stack
spec:
  replicas: 1
  # ReadWriteOnce PVC + single instance: never run two Postgres pods.
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app.kubernetes.io/name: postgres
      app.kubernetes.io/part-of: n8n-stack
  template:
    metadata:
      labels:
        app.kubernetes.io/name: postgres
        app.kubernetes.io/part-of: n8n-stack
    spec:
      securityContext:
        # postgres:18 image runs as the "postgres" user (UID 999) internally,
        # but the official image starts as root to chown the data dir.
        fsGroup: 999
        # seccompProfile:
        #   type: RuntimeDefault
      containers:
        - name: postgres
          image: postgres:18
          imagePullPolicy: IfNotPresent
          env:
            - name: POSTGRES_USER
              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_USER
            - name: POSTGRES_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_PASSWORD
            - name: POSTGRES_DB
              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_DB
            - name: POSTGRES_NON_ROOT_USER
              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_NON_ROOT_USER
            - name: POSTGRES_NON_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_NON_ROOT_PASSWORD
            # postgres:18 moved its default data dir; pin it.
            - name: PGDATA
              value: /var/lib/postgresql/data
          ports:
            - name: postgres
              containerPort: 5432
              protocol: TCP
          volumeMounts:
            - name: postgres-data
              mountPath: /var/lib/postgresql/data
            - name: initdb
              mountPath: /docker-entrypoint-initdb.d
              readOnly: true
          readinessProbe:
            exec:
              command:
                - /bin/sh
                - -c
                - 'pg_isready -h 127.0.0.1 -U "$POSTGRES_USER" -d "$POSTGRES_DB"'
            initialDelaySeconds: 5
            periodSeconds: 5
            timeoutSeconds: 5
            failureThreshold: 10
          livenessProbe:
            exec:
              command:
                - /bin/sh
                - -c
                - 'pg_isready -h 127.0.0.1 -U "$POSTGRES_USER" -d "$POSTGRES_DB"'
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 5
            failureThreshold: 6
          resources:
            requests:
              cpu: 250m
              memory: 512Mi
            limits:
              cpu: "2"
              memory: 2Gi
      volumes:
        - name: postgres-data
          persistentVolumeClaim:
            claimName: postgres-data
        - name: initdb
          configMap:
            name: postgres-initdb
            defaultMode: 0755

  strategy:
    type: Recreate   # RWO volume on /home/node/.n8n

      app.kubernetes.io/name: n8n
      app.kubernetes.io/part-of: n8n-stack

      labels:
        app.kubernetes.io/name: n8n

        - name: wait-for-postgres
          image: postgres:18
          imagePullPolicy: IfNotPresent
          command:
            - /bin/sh
            - -c
            - |
              until pg_isready -h postgres -p 5432 -U "$POSTGRES_NON_ROOT_USER" -d "$POSTGRES_DB"; do
                echo "waiting for postgres..."
                sleep 2
              done
              echo "postgres is ready"
          env:

                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_NON_ROOT_USER

              valueFrom:
                secretKeyRef:
                  name: postgres-secret
                  key: POSTGRES_DB
          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop: ["ALL"]

                  key: POSTGRES_NON_ROOT_PASSWORD

              value: external
