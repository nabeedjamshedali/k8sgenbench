```yaml
---
# Namespace
apiVersion: v1
kind: Namespace
metadata:
  name: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app

---
# Secret for Postgres credentials
apiVersion: v1
kind: Secret
metadata:
  name: postgres-secret
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
type: Opaque
stringData:
  POSTGRES_USER: "postgres"
  POSTGRES_PASSWORD: "postgres"  # Replace with a strong password / use external secret manager in real prod

---
# ConfigMap for Redis healthcheck script
apiVersion: v1
kind: ConfigMap
metadata:
  name: redis-healthcheck
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
data:
  redis.sh: |
    #!/bin/sh
    set -e
    redis-cli ping | grep -q PONG

---
# ConfigMap for Postgres healthcheck script
apiVersion: v1
kind: ConfigMap
metadata:
  name: postgres-healthcheck
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
data:
  postgres.sh: |
    #!/bin/sh
    set -e
    pg_isready -U "$POSTGRES_USER" -d postgres

---
# Redis Service (headless for stable DNS + regular for clients)
apiVersion: v1
kind: Service
metadata:
  name: redis
  namespace: voting-app
  labels:
    app.kubernetes.io/name: redis
    app.kubernetes.io/part-of: voting-app
spec:
  selector:
    app.kubernetes.io/name: redis
  ports:
    - name: redis
      port: 6379
      targetPort: 6379
  type: ClusterIP

---
# Redis Deployment
apiVersion: apps/v1
kind: Deployment
metadata:
  name: redis
  namespace: voting-app
  labels:
    app.kubernetes.io/name: redis
    app.kubernetes.io/part-of: voting-app
spec:
  replicas: 1
  selector:
    matchLabels:
      app.kubernetes.io/name: redis
  template:
    metadata:
      labels:
        app.kubernetes.io/name: redis
        app.kubernetes.io/part-of: voting-app
    spec:
      securityContext:
        runAsNonRoot: true
        runAsUser: 999
        fsGroup: 999
      containers:
        - name: redis
          image: redis:alpine
          imagePullPolicy: IfNotPresent
          ports:
            - name: redis
              containerPort: 6379
          resources:
            requests:
              cpu: 100m
              memory: 128Mi
            limits:
              cpu: 500m
              memory: 256Mi
          readinessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/redis.sh"]
            initialDelaySeconds: 5
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 3
          livenessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/redis.sh"]
            initialDelaySeconds: 15
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 3
          volumeMounts:
            - name: healthchecks
              mountPath: /healthchecks
              readOnly: true
      volumes:
        - name: healthchecks
          configMap:
            name: redis-healthcheck
            defaultMode: 0755

---
# Postgres Service
apiVersion: v1
kind: Service
metadata:
  name: db
  namespace: voting-app
  labels:
    app.kubernetes.io/name: db
    app.kubernetes.io/part-of: voting-app
spec:
  selector:
    app.kubernetes.io/name: db
  ports:
    - name: postgres
      port: 5432
      targetPort: 5432
  type: ClusterIP

---
# Postgres PersistentVolumeClaim
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: db-data
  namespace: voting-app
  labels:
    app.kubernetes.io/name: db
    app.kubernetes.io/part-of: voting-app
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi

---
# Postgres Deployment
apiVersion: apps/v1
kind: Deployment
metadata:
  name: db
  namespace: voting-app
  labels:
    app.kubernetes.io/name: db
    app.kubernetes.io/part-of: voting-app
spec:
  replicas: 1
  strategy:
    type: Recreate
  selector:
    matchLabels:
      app.kubernetes.io/name: db
  template:
    metadata:
      labels:
        app.kubernetes.io/name: db
        app.kubernetes.io/part-of: voting-app
    spec:
      securityContext:
        fsGroup: 999
      containers:
        - name: db
          image: postgres:15-alpine
          imagePullPolicy: IfNotPresent
          ports:
            - name: postgres
              containerPort: 5432
          envFrom:
            - secretRef:
                name: postgres-secret
          env:
            - name: PGDATA
              value: /var/lib/postgresql/data/pgdata
          resources:
            requests:
              cpu: 250m
              memory: 256Mi
            limits:
              cpu: "1"
              memory: 512Mi
          readinessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/postgres.sh"]
            initialDelaySeconds: 10
            periodSeconds: 5
            timeoutSeconds: 3
            failureThreshold: 5
          livenessProbe:
            exec:
              command: ["/bin/sh", "/healthchecks/postgres.sh"]
            initialDelaySeconds: 30
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 5
          volumeMounts:
            - name: db-data
              mountPath: /var/lib/postgresql/data
            - name: healthchecks
              mountPath: /healthchecks
              readOnly: true
      volumes:
        - name: db-data
          persistentVolumeClaim:
            claimName: db-data
        - name: healthchecks
          configMap:
            name: postgres-healthcheck
            defaultMode: 0755

---

# Ingress for external access to vote and result
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: voting-app
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  ingressClassName: nginx
  rules:
    - host: vote.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: vote
                port:
                  number: 80
    - host: result.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: result
                port:
                  number: 80

# NetworkPolicy: restrict Redis to only vote + worker
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: redis-allow-vote-worker
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/name: redis
  policyTypes:
    - Ingress
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app.kubernetes.io/name: vote
        - podSelector:
            matchLabels:
              app.kubernetes.io/name: worker
      ports:
        - protocol: TCP
          port: 6379

# NetworkPolicy: deny all ingress by default in namespace (allow DNS egress is implicit via kube-dns allow rule below)
apiVersion: networking.k8s.io/v1

metadata:
  name: default-deny-ingress
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
spec:
  podSelector: {}
  policyTypes:
    - Ingress

---
# NetworkPolicy: allow ingress to vote and result from ingress controller namespace
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-ingress-controller
  namespace: voting-app
  labels:
    app.kubernetes.io/part-of: voting-app
spec:
  podSelector:
    matchExpressions:
      - key: app.kubernetes.io/name
        operator: In
        values: ["vote", "result"]
  policyTypes:
    - Ingress
  ingress:
    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: ingress-nginx
      ports:
        - protocol: TCP
```
