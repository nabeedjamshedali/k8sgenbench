      app: front-end

      app: edge-router

    metadata:

    spec:
      containers:
        - name: edge-router
          image: weaveworksdemos/edge-router:0.1.1

            readOnlyRootFilesystem: true

  ports:
    - name: http
      port: 80
      targetPort: 80
      nodePort: 30001
    - name: admin
      port: 8080
      targetPort: 8080
      nodePort: 30080
